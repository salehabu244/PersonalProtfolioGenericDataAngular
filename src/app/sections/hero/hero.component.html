public class OrderService
  : IOrderService
{
  public async Task<OrderDto> GetAsync(int id)
  {
    return await _db.Orders
      .AsNoTracking()
      .Where(o => o.Id == id)
      .ProjectTo<OrderDto>()
      .SingleAsync();
  }
}
